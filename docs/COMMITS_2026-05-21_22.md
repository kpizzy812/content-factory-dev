### `GET /api/analytics/accounts-summary` — новый endpoint

`server/api/analytics/accounts-summary.get.ts` (полная имплементация):

```ts
const VALID_PLATFORMS: Platform[] = ["tiktok", "instagram", "youtube"]
const RECENT_SNAPSHOTS_LIMIT = 14

export default defineEventHandler(async (event) => {
  await requireScopedAccess(event, {
    permissions: ["canRead"],
    moduleSlug: "analytics",
  })

  const query = getQuery(event)
  const appIdRaw = Number(query.appId)
  const appIdFilter =
    Number.isFinite(appIdRaw) && appIdRaw > 0 ? appIdRaw : undefined
  const platformFilter =
    typeof query.platform === "string"
    && VALID_PLATFORMS.includes(query.platform as Platform)
      ? (query.platform as Platform)
      : undefined

  const accountWhere: Record<string, unknown> = {}
  if (appIdFilter !== undefined) accountWhere.appId = appIdFilter
  if (platformFilter !== undefined) accountWhere.platform = platformFilter

  const accounts = await prisma.socialAccount.findMany({
    where: accountWhere,
    orderBy: { createdAt: "desc" },
    select: {
      id: true,
      displayName: true,
      platform: true,
      platformHandle: true,
      status: true,
      app: { select: { id: true, name: true } },
      metricsSnapshots: {
        orderBy: { fetchedAt: "desc" },
        take: RECENT_SNAPSHOTS_LIMIT,
      },
      _count: { select: { metricsSnapshots: true } },
    },
  })

  let totalFollowersBig = 0n
  let accountsWithMetrics = 0
  let engagementSum = 0
  let engagementCount = 0

  const items = accounts.map((acc) => {
    const recentSerialized = acc.metricsSnapshots.map((s) =>
      serializeSnapshot(s, { includeRaw: false }),
    )
    const latestOk = recentSerialized.find((s) => s.status === "ok") ?? null
    const lastFetchedAt = recentSerialized[0]?.fetchedAt ?? null

    if (latestOk) {
      accountsWithMetrics += 1
      if (latestOk.followers !== null) {
        try { totalFollowersBig += BigInt(latestOk.followers) } catch {}
      }
      if (latestOk.engagementRate !== null) {
        engagementSum += latestOk.engagementRate
        engagementCount += 1
      }
    }

    return {
      account: { id: acc.id, displayName: acc.displayName, platform: acc.platform,
        platformHandle: acc.platformHandle, status: acc.status, app: acc.app },
      latestOkSnapshot: latestOk,
      recentSnapshots: recentSerialized,
      lastFetchedAt,
      snapshotsCount: acc._count.metricsSnapshots,
    }
  })

  return {
    data: {
      items,
      aggregate: {
        accountsTotal: items.length,
        accountsWithMetrics,
        totalFollowers: totalFollowersBig.toString(),
        avgEngagement:
          engagementCount > 0 ? engagementSum / engagementCount : null,
      },
    },
  }
})
```

**Ключевые детали:**
- `BigInt` агрегация (`totalFollowersBig`) и сериализация в string (Cloudflare-safe response).
- Apify-метрики — *публичный профиль* → покрывает любой `postingMethod`.
- Sparkline component сам делает фильтр `ok`-only и переворот ASC — endpoint только гарантирует DESC.

### `fillRandomTime` — anti-detect окно 1–24 ч

```ts
// Anti-detect: рандомное окно от 1 до 24 часов в будущее, шаг — минуты.
function fillRandomTime() {
  const minMs = 60 * 60 * 1000
  const maxMs = 24 * 60 * 60 * 1000
  const offset = minMs + Math.random() * (maxMs - minMs)
  const target = new Date(Date.now() + offset)
  const pad = (n: number) => String(n).padStart(2, "0")
  scheduledAtLocal.value =
    `${target.getFullYear()}-${pad(target.getMonth() + 1)}-${pad(target.getDate())}`
    + `T${pad(target.getHours())}:${pad(target.getMinutes())}`
  if (scheduledMode.value !== "scheduled") {
    scheduledMode.value = "scheduled"
  }
}
```

### Submit — hashtags parser + diagnostic mapping

```ts
async function submit() {
  if (!canSubmit.value) return
  const account = selectedAccount.value
  if (!account) return

  error.value = null
  submitting.value = true

  // Hashtags из строки: "#tag1, tag2 #tag3" → ["tag1","tag2","tag3"]
  const hashtags = hashtagsRaw.value
    .split(/[\s,]+/)
    .map((s) => s.replace(/^#/, "").trim())
    .filter((s) => s.length > 0)

  const scheduledAtIso = scheduledMode.value === "scheduled" && scheduledAtLocal.value
    ? new Date(scheduledAtLocal.value).toISOString()
    : null

  const body = {
    videoId: videoId.value!,
    socialAccountId: socialAccountId.value!,
    platform: account.platform,
    contentSnapshot: {
      caption: caption.value.trim() || null,
      description: description.value.trim() || null,
      hashtags,
    },
    scheduledAt: scheduledAtIso,
  }

  try {
    const res = await $fetch<{ data: PostingJobDto }>(
      "/api/posting-jobs",
      { method: "POST", body },
    )
    emit("created", res.data)
    close()
  } catch (e: unknown) {
    error.value = toDiagnosticError(e, {
      phase: "posting_job_create",
      url: "/api/posting-jobs",
    })
  } finally {
    submitting.value = false
  }
}
```

```ts
// Бейдж свежести: «обновлено N часов назад» / «N минут назад» / «N дней назад».
// Источник — lastFetchedAt (любой статус снапшота).
const freshness = computed(() => {
  const ts = props.item.lastFetchedAt
  if (!ts) return null
  const diffMs = Date.now() - new Date(ts).getTime()
  if (!Number.isFinite(diffMs) || diffMs < 0) return null
  const mins = Math.floor(diffMs / 60_000)
  if (mins < 1) return { label: "только что", warn: false }
  if (mins < 60) return { label: `${mins} мин назад`, warn: false }
  const hours = Math.floor(mins / 60)
  if (hours < 24) return { label: `${hours} ч назад`, warn: false }
  const days = Math.floor(hours / 24)
  // Apify-метрики собираются раз в 24 часа (idempotent), старше 2 дней —
  // повод обновить.
  return { label: `${days} д назад`, warn: days >= 2 }
})
```

```ts
/**
 * Indigo X masking flags — управляют тем, как Indigo маскирует каждый аспект
 * fingerprint'а. Семантика значений (общая для большинства flag'ов):
 *   - "natural" — Indigo не трогает, отдаёт реальные значения хоста
 *   - "mask"    — Indigo подменяет на свои masked значения (auto)
 *   - "custom"  — Indigo берёт значения из соответствующего fingerprint sub-block;
 *                 если sub-block отсутствует → 400 BAD_REQUEST_BODY.

 *   - "mask"     — Indigo cloud-proxy (если подключён)
 *   - "custom"   — наш `parameters.proxy` блок ОБЯЗАТЕЛЕН.
 */
export type IndigoMaskingMode = "natural" | "mask" | "custom"
export type IndigoProxyMaskingMode = "disabled" | "mask" | "custom"
export type IndigoGeolocationPopup = "prompt" | "allow" | "block"

export interface IndigoFlags {
  audio_masking?: IndigoMaskingMode
  fonts_masking?: IndigoMaskingMode
  geolocation_masking?: IndigoMaskingMode
  geolocation_popup?: IndigoGeolocationPopup
  graphics_masking?: IndigoMaskingMode
  graphics_noise?: IndigoMaskingMode
  localization_masking?: IndigoMaskingMode
  media_devices_masking?: IndigoMaskingMode
  navigator_masking?: IndigoMaskingMode
  ports_masking?: IndigoMaskingMode
  proxy_masking?: IndigoProxyMaskingMode
  screen_masking?: IndigoMaskingMode
  timezone_masking?: IndigoMaskingMode
  webrtc_masking?: IndigoMaskingMode
}

export interface IndigoCreateProfileParameters {
  fingerprint?: IndigoFingerprintPayload
  /**
   * Proxy лежит ВНУТРИ `parameters` (НЕ top-level body). Подтверждено через
   * официальный quick-profile docs пример. Если положить на top-level — Indigo
   * молча игнорирует и профиль ходит через хостовый IP.
   *
   * Также требует `flags.proxy_masking = "custom"` иначе Indigo не активирует
   * прокси даже при корректном расположении.
   */
  proxy?: IndigoRemoteProxy
  flags?: IndigoFlags
  storage?: { is_local?: boolean; save_service_worker?: boolean }
  [key: string]: unknown
}
```

### `buildFlags` — platform-aware

`server/utils/indigo/build-create-body.ts` (lines 330-372):

```ts
function buildFlags(opts: {
  isMobile: boolean
  hasProxy: boolean
  hasLocalization: boolean
  hasTimezone: boolean
  hasScreen: boolean
  isPartialUpdate: boolean
}): IndigoFlags {
  // На mobile все masking которые иначе были бы "natural" → "mask".
  // На desktop "natural" остаётся (Indigo принимает).
  const naturalOrMask: "natural" | "mask" = opts.isMobile ? "mask" : "natural"
  const flags: IndigoFlags = {
    // Custom где у нас есть данные:
    navigator_masking: "custom",
    localization_masking: opts.hasLocalization ? "custom" : "mask",
    timezone_masking: opts.hasTimezone ? "custom" : "mask",
    // screen — на desktop fallback "natural" (хост screen), на mobile нельзя
    // "natural" по той же причине что fonts (Linux VM screen ≠ phone screen).
    screen_masking: opts.hasScreen ? "custom" : naturalOrMask,
    // Mobile-aware: эти 5 flags на mobile = "mask", на desktop = "natural".
    audio_masking: naturalOrMask,
    fonts_masking: naturalOrMask,
    graphics_masking: naturalOrMask,
    graphics_noise: naturalOrMask,
    webrtc_masking: naturalOrMask,
    // Эти не зависят от платформы — значения как в docs quick-profile example.
    geolocation_masking: "mask",
    geolocation_popup: "prompt",
    media_devices_masking: "mask",
  }
  // proxy_masking — критичный flag для anti-leak. Без "custom" Indigo
  // игнорирует parameters.proxy полностью.
  if (!opts.isPartialUpdate) {
    flags.proxy_masking = opts.hasProxy ? "custom" : "disabled"
  }
  return flags
}
```
